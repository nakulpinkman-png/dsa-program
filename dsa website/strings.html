Program 1 - Convert String to Uppercase

#include <stdio.h> #include <ctype.h> int main() { char s[100]; printf("Enter a string: "); fgets(s, sizeof(s), stdin); for(int i = 0; s[i] != '\0'; i++) { s[i] = toupper((unsigned char)s[i]); } printf("Uppercase: %s", s); return 0; } INPUT: 
hello world
OUTPUT: 
HELLO WORLD

Program 21 - Print All Permutations of ABCDE

#include <stdio.h> void swap(char *a, char *b) { char temp = *a; *a = *b; *b = temp; } void permute(char s[], int start, int end) { if(start == end) { printf("%s\n", s); return; } for(int i = start; i <= end; i++) { swap(&s[start], &s[i]); permute(s, start + 1, end); swap(&s[start], &s[i]); } } int main() { char s[] = "ABCDE"; permute(s, 0, 4); return 0; } INPUT:

Program 22 - Sort String in Ascending Order

dcba
OUTPUT: 
Sorted string: abcd

Program 25 - Remove Leading and Trailing Spaces
