Program 4: Traverse Adjacency List with Depth First Search (DFS)

#include <stdio.h> #include <stdlib.h> struct Node { int vertex; struct Node *next; }; struct Node *adj[20]; int visited[20]; void addEdge(int u, int v) { struct Node *newNode; newNode = (struct Node *)malloc(sizeof(struct Node)); newNode->vertex = v; newNode->next = adj[u]; adj[u] = newNode; newNode = (struct Node *)malloc(sizeof(struct Node)); newNode->vertex = u; newNode->next = adj[v]; adj[v] = newNode; } void DFS(int vertex) { struct Node *temp; visited[vertex] = 1; printf("%d ", vertex); temp = adj[vertex]; while (temp != NULL) { if (visited[temp->vertex] == 0) { DFS(temp->vertex); } temp = temp->next; } } int main() { int n, e, i, u, v, start; printf("Enter number of vertices: "); scanf("%d", &n); printf("Enter number of edges: "); scanf("%d", &e); for (i = 0; i < e; i++) { printf("Enter edge (u v): "); scanf("%d %d", &u, &v); addEdge(u, v); } printf("Enter starting vertex: "); scanf("%d", &start); printf("DFS Traversal: "); DFS(start); return 0; } Input: 
5
5
0 1
0 2
1 3
2 3
3 4
0
Output: 
DFS Traversal: 0 2 3 4 1

Program 5: Traverse Adjacency Matrix with Breadth First Search (BFS)

#include <stdio.h> int graph[20][20], visited[20], queue[20]; int n, front = 0, rear = -1; void BFS(int start) { int i, vertex; visited[start] = 1; queue[++rear] = start; while (front <= rear) { vertex = queue[front++]; printf("%d ", vertex); for (i = 0; i < n; i++) { if (graph[vertex][i] == 1 && visited[i] == 0) { visited[i] = 1; queue[++rear] = i; } } } } int main() { int e, i, u, v, start; printf("Enter number of vertices: "); scanf("%d", &n); printf("Enter number of edges: "); scanf("%d", &e); for (i = 0; i < e; i++) { printf("Enter edge (u v): "); scanf("%d %d", &u, &v); graph[u][v] = 1; graph[v][u] = 1; } printf("Enter starting vertex: "); scanf("%d", &start); printf("BFS Traversal: "); BFS(start); return 0; } Input: 
5
5

Program 6: Traverse Adjacency List with Breadth First Search (BFS)
