#include <stdio.h> #define SIZE 100 int main() { int queue[SIZE]; int stack[SIZE]; int n,i; int top=-1; printf("Enter number of elements: "); scanf("%d",&n); printf("Enter queue elements:\n"); for (i=0;i<n;i++) { scanf("%d",&queue[i]); } for (i=0;i<n;i++) { stack[++top]=queue[i]; } for (i=0;i<n;i++) { queue[i]=stack[top--]; } printf("Reversed queue:\n"); for (i=0;i<n;i++) { printf("%d ",queue[i]); } return 0; } INPUT: 
Enter number of elements: 5
Enter queue elements:
10 20 30 40 50
OUTPUT: 
Reversed queue:
50 40 30 20 10

Program 20: Rotten Oranges Using Queue

#include <stdio.h> #define SIZE 100 struct cell { int row; int col; }; int main() { int a[SIZE][SIZE]; struct cell queue[SIZE*SIZE]; int r,c; int front=0; int rear=-1; int fresh=0; int minutes=0; int i,j; int row,col; int dr[4]={-1,1,0,0}; int dc[4]={0,0,-1,1}; int k; printf("Enter rows and columns: "); scanf("%d %d",&r,&c); printf("Enter matrix (0, 1, 2):\n"); for (i=0;i<r;i++) { for (j=0;j<c;j++) { scanf("%d",&a[i][j]); if (a[i][j]==2) { queue[++rear].row=i; queue[rear].col=j; } else if (a[i][j]==1) { fresh++; } } } while (front<=rear && fresh>0) { int levelEnd=rear; while (front<=levelEnd) { row=queue[front].row; col=queue[front].col; front++; for (k=0;k<4;k++) { int nr=row+dr[k]; int nc=col+dc[k]; if (nr>=0 && nr<r && nc>=0 && nc<c && a[nr][nc]==1) { a[nr][nc]=2; fresh--; queue[++rear].row=nr; queue[rear].col=nc; } } } minutes++; } if (fresh==0) { printf("Minimum time = %d minutes",minutes); } else { printf("All oranges cannot rot"); } return 0; } INPUT: 
Enter rows and columns: 3 3

Enter matrix (0, 1, 2):
2 1 1
1 1 0
0 1 1
OUTPUT: 
Minimum time = 4 minutes
