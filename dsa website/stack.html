Program 1: Create a Stack Using Integer Array (Push, Pop, Peek, IsEmpty, IsFull)

"Stack is empty\n" : "Stack is not empty\n"); break; case 5: printf(isFull() ? "Stack is full\n" : "Stack is not full\n"); break; case 6: return 0; default: printf("Invalid choice\n"); } } } INPUT:

Program 5: Enter a String from User and Reverse It Using Stack

#include <stdio.h> #include <string.h> #define SIZE 100 int main() { char str[SIZE]; char stack[SIZE]; int top = -1; int i; printf("Enter a string: "); fgets(str, SIZE, stdin); str[strcspn(str, "\n")] = '\0'; for (i = 0; str[i] != '\0'; i++) { stack[++top] = str[i]; } printf("Reversed string: "); while (top != -1) { printf("%c", stack[top--]); } return 0; } INPUT:

Program 6: Create a Program to Match Brackets Using Stack

Program 8: Sort the Stack in Ascending Order Using Integer Array

Program 9: Enter Infix Expression from User and Convert into Prefix Expression

#include <stdio.h> #include <string.h> #include <ctype.h> #define SIZE 100 char stack[SIZE]; int top = -1; int precedence(char ch) { if (ch == '^') { return 3; } if (ch == '*' || ch == '/') { return 2; } if (ch == '+' || ch == '-') { return 1; } return 0; } int main() { char infix[SIZE], prefix[SIZE]; char temp[SIZE]; int i, j = 0; char ch; printf("Enter infix expression: "); scanf("%99s", infix); strcpy(temp, infix); strrev(temp); for (i = 0; temp[i] != '\0'; i++) { if (temp[i] == '(') { temp[i] = ')'; } else if (temp[i] == ')') { temp[i] = '('; } } for (i = 0; temp[i] != '\0'; i++) { ch = temp[i]; if (isalnum(ch)) { prefix[j++] = ch; } else if (ch == '(') { stack[++top] = ch; } else if (ch == ')') { while (top != -1 && stack[top] != '(') { prefix[j++] = stack[top--]; } if (top != -1) { top--; } } else { while (top != -1 && stack[top] != '(' && precedence(stack[top]) > precedence(ch)) { prefix[j++] = stack[top--]; } stack[++top] = ch; } } while (top != -1) { prefix[j++] = stack[top--]; } prefix[j] = '\0'; strrev(prefix); printf("Prefix expression: %s", prefix); return 0; } INPUT: 
Enter infix expression: A+B*C
OUTPUT: 
Prefix expression: +A*BC
Note: This program uses `strrev()`, which is supported by some compilers but is not part of standard C. If your compiler does not support it, use a manual string-reversal function.

Program 13: Implement Tower of Hanoi Using Stacks

#include <stdio.h> #define SIZE 100 int stack[3][SIZE]; int top[3] = {-1, -1, -1}; char rodName[3] = {'A', 'B', 'C'}; void push(int rod, int disk) { stack[rod][++top[rod]] = disk; } int pop(int rod) { if (top[rod] == -1) { return 0; } return stack[rod][top[rod]--]; } void moveDisk(int from, int to) { int a = pop(from); int b = pop(to); if (a == 0) { push(from, b); printf("Move disk %d from %c to %c\n", b, rodName[to], rodName[from]); } else if (b == 0) { push(to, a); printf("Move disk %d from %c to %c\n", a, rodName[from], rodName[to]); } else if (a > b) { push(from, a); push(from, b); printf("Move disk %d from %c to %c\n", b, rodName[to], rodName[from]); } else { push(to, b); push(to, a); printf("Move disk %d from %c to %c\n", a, rodName[from], rodName[to]); } } int main() { int n, i; int totalMoves; int from, to; printf("Enter number of disks: "); scanf("%d", &n); if (n < 1 || n >= SIZE) { printf("Invalid number of disks"); return 0; } for (i = n; i >= 1; i--) { push(0, i); } totalMoves = (1 << n) - 1; if (n % 2 == 0) { rodName[1] = 'C'; rodName[2] = 'B'; } for (i = 1; i <= totalMoves; i++) { if (i % 3 == 1) { from = 0; to = 2; } else if (i % 3 == 2) { from = 0; to = 1; } else { from = 1; to = 2; } moveDisk(from, to); } return 0; } INPUT: 
Enter number of disks: 3
OUTPUT: 
Move disk 1 from A to C
Move disk 2 from A to B
Move disk 1 from C to B
Move disk 3 from A to C
Move disk 1 from B to A
Move disk 2 from B to C
Move disk 1 from A to C
