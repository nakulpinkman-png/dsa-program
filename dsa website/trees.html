Program 1: Construct Binary Tree from Preorder and Inorder Traversal

#include <stdio.h> #include <stdlib.h> int pre[100], in[100], p; typedef struct N { int d; struct N *l, *r; } N; N *build(int a, int b) { if (a > b) return NULL; N *t = malloc(sizeof(N)); t->d = pre[p++]; int k = a; while (in[k] != t->d) k++; t->l = build(a, k - 1); t->r = build(k + 1, b); return t; } void post(N *t) { if (t) { post(t->l); post(t->r); printf("%d ", t->d); } } int main() { int n, i; scanf("%d", &n); for (i = 0; i < n; i++) scanf("%d", &pre[i]); for (i = 0; i < n; i++) scanf("%d", &in[i]); p = 0; post(build(0, n - 1)); return 0; } INPUT: 
5
1 2 4 5 3
4 2 5 1 3
OUTPUT: 
4 5 2 3 1

Program 9: Find Inorder Predecessor and Inorder Successor of BST
