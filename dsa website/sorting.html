Program 7: Merge Sort Through Iteration on Integer Array

#include <stdio.h> int main() { int a[100], temp[100]; int n, i, width, left, mid, right; int x, y, k, start; printf("Enter array size: "); scanf("%d", &n); printf("Enter array elements:\n"); for (i = 0; i < n; i++) { scanf("%d", &a[i]); } for (width = 1; width < n; width *= 2) { for (left = 0; left < n; left += 2 * width) { mid = left + width; right = left + 2 * width; if (mid > n) { mid = n; } if (right > n) { right = n; } x = left; y = mid; k = left; while (x < mid && y < right) { if (a[x] <= a[y]) { temp[k++] = a[x++]; } else { temp[k++] = a[y++]; } } while (x < mid) { temp[k++] = a[x++]; } while (y < right) { temp[k++] = a[y++]; } for (start = left; start < right; start++) { a[start] = temp[start]; } } } printf("Sorted array:\n"); for (i = 0; i < n; i++) { printf("%d ", a[i]); } return 0; } Input: 
5

Program 8: Merge Sort Through Recursion on Integer Array

#include <stdio.h> void merge(int a[], int low, int mid, int high) { int temp[100]; int i = low, j = mid + 1, k = low; while (i <= mid && j <= high) { if (a[i] <= a[j]) { temp[k++] = a[i++]; } else { temp[k++] = a[j++]; } } while (i <= mid) { temp[k++] = a[i++]; } while (j <= high) { temp[k++] = a[j++]; } for (i = low; i <= high; i++) { a[i] = temp[i]; } } void mergeSort(int a[], int low, int high) { int mid; if (low < high) { mid = (low + high) / 2; mergeSort(a, low, mid); mergeSort(a, mid + 1, high); merge(a, low, mid, high); } } int main() { int a[100], n, i; printf("Enter array size: "); scanf("%d", &n); printf("Enter array elements:\n"); for (i = 0; i < n; i++) { scanf("%d", &a[i]); } mergeSort(a, 0, n - 1); printf("Sorted array:\n"); for (i = 0; i < n; i++) { printf("%d ", a[i]); } return 0; } Input:

Program 9: Quick Sort Through Iteration on Integer Array

#include <stdio.h> void quickSort(int a[], int n) { int low[100], high[100]; int top = -1; int l, h, i, j, pivot, temp; low[++top] = 0; high[top] = n - 1; while (top >= 0) { l = low[top]; h = high[top--]; pivot = a[h]; i = l - 1; for (j = l; j < h; j++) { if (a[j] < pivot) { i++; temp = a[i]; a[i] = a[j]; a[j] = temp; } } temp = a[i + 1]; a[i + 1] = a[h]; a[h] = temp; j = i + 1; if (j - 1 > l) { low[++top] = l; high[top] = j - 1; } if (j + 1 < h) { low[++top] = j + 1; high[top] = h; } } } int main() { int a[100], n, i; printf("Enter array size: "); scanf("%d", &n); printf("Enter array elements:\n"); for (i = 0; i < n; i++) { scanf("%d", &a[i]); } quickSort(a, n); printf("Sorted array:\n"); for (i = 0; i < n; i++) { printf("%d ", a[i]); } return 0; } Input: 
5
10 7 8 9 1
Output: 
Sorted array:
1 7 8 9 10
