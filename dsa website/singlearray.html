Program 1: Display all array values

Program 13: Count duplicate values

Count how many times duplicate values occur.

#include <stdio.h> int main() { int a[100]; int n; int i; int j; int k; int count; int seen; int total = 0; printf("Enter number of elements: "); scanf("%d", &n); printf("Enter array elements:\n"); for (i = 0; i < n; i++) { scanf("%d", &a[i]); } for (i = 0; i < n; i++) { seen = 0; for (k = 0; k < i; k++) { if (a[k] == a[i]) { seen = 1; break; } } if (seen == 1) { continue; } count = 1; for (j = i + 1; j < n; j++) { if (a[i] == a[j]) { count++; } } if (count > 1) { printf("%d occurs %d times\n", a[i], count); total = total + count - 1; } } printf("Total duplicate occurrences = %d", total); return 0; } Input 
Enter number of elements: 7
2 5 3 2 5 8 2
Output 
2 occurs 3 times
5 occurs 2 times
Total duplicate occurrences = 3

Program 17: Merge two sorted arrays

Merge two sorted arrays into one sorted array.

#include <stdio.h> int main() { int a[100]; int b[100]; int c[200]; int n; int m; int i = 0; int j = 0; int k = 0; printf("Enter size of first array: "); scanf("%d", &n); printf("Enter first sorted array:\n"); for (i = 0; i < n; i++) { scanf("%d", &a[i]); } printf("Enter size of second array: "); scanf("%d", &m); printf("Enter second sorted array:\n"); for (j = 0; j < m; j++) { scanf("%d", &b[j]); } i = 0; j = 0; while (i < n && j < m) { if (a[i] <= b[j]) { c[k] = a[i]; i++; } else { c[k] = b[j]; j++; } k++; } while (i < n) { c[k] = a[i]; i++; k++; } while (j < m) { c[k] = b[j]; j++; k++; } printf("Merged array: "); for (i = 0; i < k; i++) { printf("%d ", c[i]); } return 0; } Input 
Enter size of first array: 4
1 3 5 7
Enter size of second array: 3
2 4 6
Output 
Merged array: 1 2 3 4 5 6 7

Program 20: Trapping Rain Water Problem

Calculate how much rainwater can be trapped between the bars of an elevation map.

#include <stdio.h> int main() { int a[100]; int leftMax[100]; int rightMax[100]; int n; int i; int water = 0; printf("Enter number of bars: "); scanf("%d", &n); printf("Enter bar heights:\n"); for (i = 0; i < n; i++) { scanf("%d", &a[i]); } if (n < 3) { printf("Trapped water = 0"); return 0; } leftMax[0] = a[0]; for (i = 1; i < n; i++) { if (a[i] > leftMax[i - 1]) { leftMax[i] = a[i]; } else { leftMax[i] = leftMax[i - 1]; } } rightMax[n - 1] = a[n - 1]; for (i = n - 2; i >= 0; i--) { if (a[i] > rightMax[i + 1]) { rightMax[i] = a[i]; } else { rightMax[i] = rightMax[i + 1]; } } for (i = 0; i < n; i++) { if (leftMax[i] < rightMax[i]) { water = water + leftMax[i] - a[i]; } else { water = water + rightMax[i] - a[i]; } } printf("Trapped water = %d units", water); return 0; } Input 
Enter number of bars: 6
4 2 0 3 2 5
Output 
Trapped water = 9 units
